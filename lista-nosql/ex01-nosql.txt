db.createCollection("alunos",{
  validator: { $jsonSchema: {
    bsonType: "object",
    required: ["nome", "idade","curso"],
    properties:{
      nome: {bsonType: "string"},
      idade: {bsonType: "int", minimum: 0},
      curso: {bsonType: "string"}
    }
  }}
})

db.runCommand({
  collMod: "alunos",
  validator: { $jsonSchema: {
    bsonType: "object",
    required: ["nome", "idade","curso"],
    properties:{
      nome: {bsonType: "string"},
      idade: {bsonType: "int", minimum: 0},
      curso: {bsonType: "string"}
    }
  }},
  validationLevel: "moderate",
  validationAction: "warn"
})
