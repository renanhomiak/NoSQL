-- Aula prática: do modelo relacional ao MongoDB
-- SGBD: PostgreSQL

CREATE TABLE clientes (
  id_cliente INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  nome VARCHAR(100) NOT NULL,
  email VARCHAR(120) NOT NULL UNIQUE,
  telefone VARCHAR(20),
  ativo BOOLEAN NOT NULL DEFAULT TRUE
);

CREATE TABLE enderecos (
  id_endereco INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  id_cliente INTEGER NOT NULL REFERENCES clientes(id_cliente),
  apelido VARCHAR(30) NOT NULL,
  logradouro VARCHAR(120) NOT NULL,
  numero VARCHAR(10) NOT NULL,
  cidade VARCHAR(80) NOT NULL,
  uf CHAR(2) NOT NULL,
  cep CHAR(8) NOT NULL,
  UNIQUE (id_cliente, apelido)
);

CREATE TABLE categorias (
  id_categoria INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  nome VARCHAR(60) NOT NULL UNIQUE
);

CREATE TABLE produtos (
  id_produto INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  id_categoria INTEGER NOT NULL REFERENCES categorias(id_categoria),
  sku VARCHAR(20) NOT NULL UNIQUE,
  nome VARCHAR(120) NOT NULL,
  preco NUMERIC(10,2) NOT NULL CHECK (preco > 0),
  ativo BOOLEAN NOT NULL DEFAULT TRUE
);

CREATE TABLE pedidos (
  id_pedido INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  id_cliente INTEGER NOT NULL REFERENCES clientes(id_cliente),
  id_endereco_entrega INTEGER NOT NULL REFERENCES enderecos(id_endereco),
  data_pedido TIMESTAMP NOT NULL,
  status VARCHAR(20) NOT NULL CHECK (status IN ('CRIADO','PAGO','ENVIADO','ENTREGUE','CANCELADO')),
  frete NUMERIC(10,2) NOT NULL CHECK (frete >= 0)
);

CREATE TABLE itens_pedido (
  id_pedido INTEGER NOT NULL REFERENCES pedidos(id_pedido),
  id_produto INTEGER NOT NULL REFERENCES produtos(id_produto),
  quantidade INTEGER NOT NULL CHECK (quantidade > 0),
  preco_unitario NUMERIC(10,2) NOT NULL CHECK (preco_unitario > 0),
  PRIMARY KEY (id_pedido, id_produto)
);

--3
SELECT p.*
 FROM pedidos p
INNER JOIN enderecos e
 ON p.id_endereco_entrega = e.id_endereco
WHERE e.uf = 'SP';

--4
SELECT p.*
 FROM pedidos p
INNER JOIN clientes c 
 ON p.id_cliente = c.id_cliente
WHERE c.nome = 'Ana Souza';

--5
SELECT p.*
 FROM pedidos p
INNER JOIN itens_pedido it ON p.id_pedido = it.id_pedido
INNER JOIN produtos pro ON it.id_produto = pro.id_produto
INNER JOIN categorias c ON pro.id_categoria = c.id_categoria
WHERE c.nome = 'Livros';

--6
SELECT p.*
FROM pedidos p
JOIN itens_pedido it ON p.id_pedido = it.id_pedido
JOIN produtos pr ON it.id_produto = pr.id_produto
WHERE pr.sku = 'INF-002' AND it.quantidade >= 2;

--7
SELECT 
 p.id_pedido,
 SUM(it.quantidade * it.preco_unitario) AS total_itens,
 p.frete,
 SUM(it.quantidade * it.preco_unitario) + p.frete AS total_pedido
FROM pedidos p
JOIN itens_pedido it ON p.id_pedido = it.id_pedido
GROUP BY p.id_pedido, p.frete
ORDER BY p.id_pedido;
